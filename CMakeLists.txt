cmake_minimum_required(VERSION 3.25)
project(AC4Native VERSION 0.3.1 LANGUAGES CXX RC)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/json.hpp")
  file(DOWNLOAD
    "https://raw.githubusercontent.com/nlohmann/json/v3.12.0/single_include/nlohmann/json.hpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor/json.hpp"
    EXPECTED_HASH SHA256=aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63
    TLS_VERIFY ON)
endif()
file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/distribution/payload.zip"
     DESTINATION "${CMAKE_CURRENT_SOURCE_DIR}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/payload-hashes.json" CATALOG)
string(JSON FILE_COUNT LENGTH "${CATALOG}")
math(EXPR LAST_FILE "${FILE_COUNT} - 1")
foreach(INDEX RANGE 0 ${LAST_FILE})
  string(JSON RELATIVE_PATH MEMBER "${CATALOG}" ${INDEX})
  string(JSON EXPECTED GET "${CATALOG}" "${RELATIVE_PATH}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/payload/${RELATIVE_PATH}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Payload hash mismatch: ${RELATIVE_PATH}")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(engine STATIC engine.cpp)
target_compile_definitions(engine PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(engine PRIVATE /W4 /utf-8 /permissive- /guard:cf)
target_link_libraries(engine PUBLIC bcrypt shell32 ole32)
add_executable(AC4-PhysX-Fix-Setup WIN32 main.cpp generated/package.rc)
target_compile_options(AC4-PhysX-Fix-Setup PRIVATE /W4 /utf-8 /permissive- /guard:cf)
target_link_options(AC4-PhysX-Fix-Setup PRIVATE /DYNAMICBASE /NXCOMPAT /GUARD:CF /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
target_link_libraries(AC4-PhysX-Fix-Setup PRIVATE engine gdiplus comctl32 shlwapi)
add_executable(engine-tests tests.cpp)
target_compile_options(engine-tests PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(engine-tests PRIVATE engine)
enable_testing()
add_test(NAME transaction-tests COMMAND engine-tests)
add_executable(render-tests WIN32 main.cpp generated/package.rc)
target_compile_definitions(render-tests PRIVATE AC4_RENDER_TEST)
target_compile_options(render-tests PRIVATE /W4 /utf-8 /permissive- /guard:cf)
target_link_options(render-tests PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
target_link_libraries(render-tests PRIVATE engine gdiplus comctl32 shlwapi)
